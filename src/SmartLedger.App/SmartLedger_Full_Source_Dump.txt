using System.Linq;
using System.IO;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using SmartLedger.Core.Abstractions;

namespace SmartLedger.App.ViewModels;

public partial class BackupViewModel : ObservableObject
{
    private readonly IBackupService _backupService;
    [ObservableProperty] private string destination;
    [ObservableProperty] private string status;
    [ObservableProperty] private bool isBusy;

    public BackupViewModel(IBackupService backupService)
    {
        _backupService = backupService;
        destination = backupService.FindRecommendedDirectory() ?? "";
        status = string.IsNullOrWhiteSpace(destination)
            ? "لم يتم العثور على قرص ثانٍ مناسب. اختر قرصًا مختلفًا عن قرص النظام."
            : "المسار المقترح جاهز للنسخ الاحتياطي.";
    }

    public bool HasSecondaryDrive => !string.IsNullOrWhiteSpace(Destination) && IsSecondary(Destination);

    [RelayCommand]
    private void Choose()
    {
        var dialog = new Microsoft.Win32.OpenFolderDialog { Title = "اختر مجلد النسخ الاحتياطي" };
        if (dialog.ShowDialog() != true) return;
        Destination = dialog.FolderName;
        Status = HasSecondaryDrive
            ? "المسار صالح: قرص مختلف عن قرص النظام."
            : "هذا المجلد على قرص النظام. اختر قرصًا آخر.";
        OnPropertyChanged(nameof(HasSecondaryDrive));
        CreateCommand.NotifyCanExecuteChanged();
    }

    [RelayCommand(CanExecute = nameof(CanBackup))]
    private async Task CreateAsync()
    {
        IsBusy = true;
        Status = "جارٍ إنشاء النسخة الاحتياطية…";
        try
        {
            var result = await _backupService.CreateAsync(Destination);
            Status = $"تم الحفظ بأمان على قرص مستقل • {Path.GetFileName(result.FilePath)}";
        }
        catch (Exception ex) { Status = ex.Message; }
        finally { IsBusy = false; CreateCommand.NotifyCanExecuteChanged(); }
    }

    private bool CanBackup() => !IsBusy && HasSecondaryDrive;
    private static bool IsSecondary(string path)
    {
        var system = Path.GetPathRoot(Environment.SystemDirectory);
        var target = Path.GetPathRoot(Path.GetFullPath(path));
        return !string.Equals(system, target, StringComparison.OrdinalIgnoreCase);
    }
}

public partial class LoginViewModel : ObservableObject
{
    private readonly UserRepository _users;
    private readonly PasswordHasher _hasher;
    private readonly Action _completed;

    [ObservableProperty] private string userName = "";
    [ObservableProperty] private string password = "";
    [ObservableProperty] private string error = "";

    public LoginViewModel(UserRepository users, PasswordHasher hasher, Action completed)
        => (_users, _hasher, _completed) = (users, hasher, completed);

    [RelayCommand]
    private async Task LoginAsync()
    {
        Error = "";
        var user = await _users.FindAsync(UserName.Trim());
        if (user is null || !user.IsActive || !_hasher.Verify(Password, user.PasswordHash))
        {
            Error = "اسم المستخدم أو كلمة المرور غير صحيحة.";
            Password = "";
            return;
        }
        _completed();
    }

using System.Globalization;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using SmartLedger.Core.Abstractions;
using SmartLedger.Core.Models;

namespace SmartLedger.App.ViewModels;

public partial class OperationsViewModel(IAppRepository repo, IAppSession session) : ObservableObject
{
    [ObservableProperty] private IReadOnlyList<PersonSummary> people = Array.Empty<PersonSummary>();
    [ObservableProperty] private IReadOnlyList<LedgerOperation> operations = Array.Empty<LedgerOperation>();
    [ObservableProperty] private Guid? selectedPersonId;
    [ObservableProperty] private OperationType selectedType = OperationType.ReceiveFromPerson;
    [ObservableProperty] private string amountText = "";
    [ObservableProperty] private string note = "";
    [ObservableProperty] private string dateText = FormatDate(DateTimeOffset.Now);
    [ObservableProperty] private bool isBusy;

    public Array Types => Enum.GetValues<OperationType>();

    partial void OnSelectedPersonIdChanged(Guid? value) => SaveCommand.NotifyCanExecuteChanged();
    partial void OnAmountTextChanged(string value) => SaveCommand.NotifyCanExecuteChanged();
    partial void OnIsBusyChanged(bool value) => SaveCommand.NotifyCanExecuteChanged();

    public async Task LoadAsync()
    {
        People = await repo.GetPeopleAsync();
        Operations = await repo.GetOperationsAsync(take: 100);
        DateText = FormatDate(DateTimeOffset.Now);
    }

    [RelayCommand(CanExecute = nameof(CanSave))]
    private async Task SaveAsync()
    {
        if (!TryParseAmount(AmountText, out var amount) || SelectedPersonId is null) return;
        IsBusy = true;
        try
        {
            await repo.CreateOperationAsync(new LedgerOperation(
                Guid.NewGuid(), SelectedPersonId.Value, SelectedType, amount,
                Note.Trim(), DateTimeOffset.Now, session.CurrentUser!.Id));

            AmountText = "";
            Note = "";
            DateText = FormatDate(DateTimeOffset.Now);
            Operations = await repo.GetOperationsAsync(take: 100);
        }
        finally { IsBusy = false; }
    }

    private bool CanSave() => !IsBusy && SelectedPersonId is not null && TryParseAmount(AmountText, out var amount) && amount > 0;

    private static bool TryParseAmount(string? value, out decimal amount)
    {
        var normalized = (value ?? "").Trim().Replace(",", "").Replace("٬", "");
        return decimal.TryParse(normalized, NumberStyles.Number, CultureInfo.InvariantCulture, out amount);
    }

    private static string FormatDate(DateTimeOffset value) => value.ToLocalTime().ToString("dddd • dd/MM/yyyy • hh:mm tt", CultureInfo.CurrentCulture);
}

using SmartLedger.Core.Models;
using SmartLedger.Core.Services;
using SmartLedger.Infrastructure.Data;

namespace SmartLedger.App.ViewModels;

public partial class SetupViewModel : ObservableObject
{
    private readonly UserRepository _users;
    private readonly PasswordHasher _hasher;
    private readonly Action _completed;

    [ObservableProperty] private string userName = "";
    [ObservableProperty] private string password = "";
    [ObservableProperty] private string confirmation = "";
    [ObservableProperty] private string recoveryPhrase = "";
    [ObservableProperty] private string recoveryHint = "";
    [ObservableProperty] private string error = "";

    public SetupViewModel(UserRepository users, PasswordHasher hasher, Action completed)
        => (_users, _hasher, _completed) = (users, hasher, completed);

    public bool CanCreate => UserName.Trim().Length >= 3 && Password.Length >= 8 && Password == Confirmation && RecoveryPhrase.Trim().Length >= 8 && RecoveryHint.Trim().Length >= 2;

    partial void OnUserNameChanged(string value) => OnPropertyChanged(nameof(CanCreate));
    partial void OnPasswordChanged(string value) { OnPropertyChanged(nameof(CanCreate)); OnPropertyChanged(nameof(PasswordMatch)); }
    partial void OnConfirmationChanged(string value) { OnPropertyChanged(nameof(CanCreate)); OnPropertyChanged(nameof(PasswordMatch)); }
    partial void OnRecoveryPhraseChanged(string value) => OnPropertyChanged(nameof(CanCreate));
    partial void OnRecoveryHintChanged(string value) => OnPropertyChanged(nameof(CanCreate));

    public bool PasswordMatch => string.IsNullOrEmpty(Confirmation) || Password == Confirmation;

    [RelayCommand(CanExecute = nameof(CanCreate))]
    private async Task CreateAsync()
    {
        Error = "";
        try
        {
            await _users.CreateAdministratorAsync(new AdminSetup(UserName.Trim(), _hasher.Hash(Password), _hasher.Hash(RecoveryPhrase.Trim()), RecoveryHint.Trim()));
            _completed();
        }
        catch (Exception ex)
        {
            Error = $"تعذر إنشاء حساب الإدارة: {ex.Message}";
        }
    }
}

namespace SmartLedger.App.Views;

public partial class BootstrapWindow : Window
{
    private readonly Database _database = new(LocalAppPaths.Root);

    public BootstrapWindow()
    {
        InitializeComponent();
        Loaded += async (_, _) => await InitializeAsync();
    }

    private async Task InitializeAsync()
    {
        await _database.InitializeAsync();
        var users = new UserRepository(_database);
        if (await users.HasAdministratorAsync())
        {
            OpenLogin(users);
            return;
        }
        DataContext = new SetupViewModel(users, new PasswordHasher(), OpenLogin);
        UserNameBox.Focus();
    }

    private void PasswordChanged(object sender, RoutedEventArgs e) { if (DataContext is SetupViewModel vm) vm.Password = PasswordBox.Password; }
    private void ConfirmChanged(object sender, RoutedEventArgs e) { if (DataContext is SetupViewModel vm) vm.Confirmation = ConfirmBox.Password; }

    private void OpenLogin() => OpenLogin(new UserRepository(_database));

    private void OpenLogin(UserRepository users)
    {
        new LoginWindow(_database).Show();
        Close();
    }
}
